Basic

Two points: one at (x, y) and one at (y, x). Drag one and the other mirrors it.

const render = g9(function (params) {
  const xy = params.xy;
  const flip = np.array([[0,1],[1,0]]);
  const p2 = np.dot(flip, xy);
  return {
    p1: point(xy),
    p2: point(p2),
  };
}, { xy: [40, 0] });

Lines with affects

Three lines with different affects constraints controlling which endpoints can move.

const lines = g9(function (params) {
  const opts = { "stroke-width": 8 };
  return {
    l1: line(params.line1, { stroke: 'black', ...opts }),
    l2: line(params.line2, { stroke: '#e11d48',
         affects: { line2: [1,1,0,0] }, ...opts }),

         affects: { line3: [1,0,0,1] }, ...opts }),
  };

     line2: [-100,0,100,0],

Dragon Curve

A recursive dragon curve. Drag the endpoints or the curve to change its squareness.

const dragon = g9(function (params) {
  const points = {};
  function curve(from, to, dir, level, name) {
    if (level === 0) {
      points["ln"+name] = line(np.concatenate([from,to]));
    } else {
      const rot = np.array([[0,1],[-1,0]]);
      const mid = from.ref.add(to.ref)
        .add(np.dot(rot,to.ref.sub(from.ref))
        .mul(params.squareness).mul(dir))
        .div(2);
      curve(from, mid.ref, -1, level-1, name+"l");
      curve(mid, to, 1, level-1, name+"r");
    }
  }
  curve(params.fromPt, params.toPt, -1, 5, "");
  points.from = point(params.fromPt.ref);
  points.to = point(params.toPt.ref);
  return points;
}, { fromPt:[175,96], toPt:[-175,39], squareness:[0.8] });

Fractal Tree

A recursive tree. Drag branches to change the split angle and attenuation.

const tree = g9(function (params) {
  const points = {};
  function branch(pt, len, angle, n, name) {
    const rad = angle.mul(Math.PI / 180);
    const dir = np.stack([np.cos(rad.ref), np.sin(rad)]);
    const next = pt.ref.add(dir.mul(len.ref));
    points["pt"+name] = point(next.ref, {fill:"green"});
    points["ln"+name] = line(np.concatenate([pt,next.ref]),
      { stroke: n > 3 ? "brown" : "green",
        affects: {deltaAngle:true, attenuation:true},
        "stroke-width": 3 });
    if (n > 0) {
      branch(next.ref, len.mul(params.attenuation.ref),
        angle.ref.add(params.deltaAngle.ref), n-1, name+"l");
      branch(next, len.mul(params.attenuation),
        angle.sub(params.deltaAngle), n-1, name+"r");
    }
  }
  branch(...);
  return points;
}, { deltaAngle:[33], startLength:[75], attenuation:[0.7] });
